2. Regression Model in Keras


import numpy as np

from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Dense

X = np.array([[25,2],[30,4],[35,5],[40,7]], dtype=float)

y = np.array([2000,3000,4000,5000], dtype=float)

model = Sequential([
    Dense(4, activation='relu', input_shape=(2,)),
    Dense(1)
])

model.compile(optimizer='adam', loss='mse')

model.fit(X, y, epochs=100, verbose=0)

print(model.predict(np.array([[38,6]])))

5. CNN Image Classifier


import tensorflow as tf

import matplotlib.pyplot as plt

from tensorflow.keras import layers, models

(X, y), _ = tf.keras.datasets.mnist.load_data()

plt.imshow(X[0], cmap='gray')
plt.show()

model = models.Sequential([

    layers.Conv2D(
        32,
        (3,3),
        activation='relu',
        input_shape=(28,28,1)
    ),

    layers.Flatten(),

    layers.Dense(10, activation='softmax')

])

print("CNN Model Ready")

6. Hyperparameter Tuning


import tensorflow as tf

from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Dense

import numpy as np

X = np.array([[0],[1],[2],[3]], dtype=float)

y = np.array([[0],[2],[4],[6]], dtype=float)

model = Sequential([
    Dense(16, activation='relu', input_shape=(1,)),
    Dense(1)
])

optimizer = tf.keras.optimizers.Adam(
    learning_rate=0.01
)

model.compile(
    optimizer=optimizer,
    loss='mse'
)

model.fit(X, y, epochs=200, verbose=0)

print(model.predict(np.array([[5]])))

9. Sentiment Analysis using RNN


import numpy as np

from tensorflow.keras.models import Sequential

from tensorflow.keras.layers import (
    Embedding,
    SimpleRNN,
    Dense
)

from tensorflow.keras.preprocessing.sequence import pad_sequences

X = [[1,2,3],[2,3,4],[1,4,5]]

y = np.array([1,0,1])

X = pad_sequences(X, maxlen=5)

model = Sequential([

    Embedding(
        input_dim=10,
        output_dim=8,
        input_length=5
    ),

    SimpleRNN(8),

    Dense(1, activation='sigmoid')

])

model.compile(
    optimizer='adam',
    loss='binary_crossentropy',
    metrics=['accuracy']
)

model.fit(X, y, epochs=10, verbose=0)

print(model.predict(X))

10. LSTM Autoencoder


import numpy as np

from tensorflow.keras.models import Sequential

from tensorflow.keras.layers import (
    LSTM,
    RepeatVector,
    TimeDistributed,
    Dense
)

X = np.random.rand(10,5,1)

model = Sequential([

    LSTM(8, input_shape=(5,1)),

    RepeatVector(5),

    LSTM(8, return_sequences=True),

    TimeDistributed(Dense(1))

])

model.compile(
    optimizer='adam',
    loss='mse'
)

model.fit(X, X, epochs=10, verbose=0)

print("LSTM Autoencoder Ready")
