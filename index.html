11. Raspberry Pi Sensor

from machine import Pin,time_pulse_us
import time
trig=Pin(3,Pin.OUT)
echo=Pin(2,Pin.IN)
while True:
 trig.low()
 time.sleep_us(2)
 trig.high()
 time.sleep_us(10)
 trig.low()
 duration=time_pulse_us(echo,1)
 distance=duration/58
 print("Distance:",distance,"cm")
 time.sleep(1)

12. Cloud Data Logging
