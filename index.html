7. Logical Operations

{

 while(1);
}

8. UART Communication

#include 

void UART_Init()
{
 TMOD = 0x20;
 TH1 = 0xFD;
 SCON = 0x50;
 TR1 = 1;
}

void UART_Send(char ch)
{
 SBUF = ch;
 while(TI == 0);
 TI = 0;
}

void main()
{
 UART_Init();
 UART_Send('H');
 UART_Send('I');
 while(1);
}

11. Raspberry Pi Sensor

from machine import Pin, time_pulse_us
import time

trig = Pin(3, Pin.OUT)
echo = Pin(2, Pin.IN)

while True:
 trig.low()
 time.sleep_us(2)
 trig.high()
 time.sleep_us(10)
 trig.low()

 duration = time_pulse_us(echo, 1)
 distance = duration / 58

 print("Distance:", distance, "cm")
 time.sleep(1)
